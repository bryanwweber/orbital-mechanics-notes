# Planetary Parameters

Further information about the [planetary physical parameters](https://ssd.jpl.nasa.gov/planets/phys_par.html), including possibly updated values, is available from the NASA Jet Propulsion Laboratory Solar System Dynamics group. Data on this page were generated on {eval}`today`.

## Planetary Mass Parameters

The data in @tab:planetary-mass-parameters is generated by the following script, using data from @Park2021:

```{code-cell} python
:tags: ["hide-cell"]
:label: code:planetary-mass-parameters
from io import StringIO

import pandas as pd
import pint_pandas
from pint import UnitRegistry, set_application_registry

units = UnitRegistry()
set_application_registry(units)

object_identifier_map = {
    "GM1": "Mercury",
    "GM2": "Venus",
    "GM3": "Earth",
    "GM4": "Mars",
    "GM5": "Jupiter",
    "GM6": "Saturn",
    "GM7": "Uranus",
    "GM8": "Neptune",
    "GM9": "Pluto",
    "GMS": "Sun",
    "GMM": "Moon",
    "GMB": "Earth-Moon-Barycenter",
}

# https://naif.jpl.nasa.gov/pub/naif/generic_kernels/spk/planets/de442_tech-comments.txt
# Mass parameter (GM) for Sun, Moon, and Planets:
#           AU**3/DAY**2                    GMSun/GM(I)         KM**3/SEC**2
data = """\
Celestial Body;GM_in_au;GM_ratio;GM_in_km
GMS;2.9591220828406134e-04;1.000000;132712440041.253311
GM1;4.9125001948893182e-11;6023657.944928;22031.868551
GM2;7.2434523326441187e-10;408523.718656;324858.592000
GM3;8.8876924475211348e-10;332946.048742;398600.435544
GM4;9.5495488297258119e-11;3098703.546737;42828.375816
GM5;2.8253458252257917e-07;1047.348631;126712764.100000
GM6;8.4597059933762903e-08;3497.901801;37940584.841800
GM7;1.2920265649682399e-08;22902.950783;5794556.400000
GM8;1.5243573478851939e-08;19412.259776;6836527.100580
GM9;2.1750964648933581e-12;136045556.167353;975.500000
GMM;1.0931894592103165e-11;27068703.031387;4902.800104
GMB;8.9970113934421660e-10;328900.559690;403503.235648
"""

raw_gm_data = (
    pd.read_csv(
        StringIO(data),
        sep=";",
        header=0,
        index_col=0,
        engine="c",
        converters={0: lambda x: object_identifier_map[x]},
    )
    .drop(labels=["Moon", "Earth-Moon-Barycenter"], axis=0)
)

gm_data = pd.DataFrame(
    {
        "GM_in_km": pint_pandas.PintArray(
            raw_gm_data["GM_in_km"].values, dtype="pint[km**3/s**2]"
        ),
    },
    index=raw_gm_data.index,
)

# Pint uses the CODATA-2022 value for G released in May 2024, the most current value as
# of 09-FEB-2025
gm_data["mass"] = (gm_data["GM_in_km"] / units.gravitational_constant).pint.to("kg")
gm_data["Mass Fraction [kg/kg]"] = gm_data["mass"] / gm_data["mass"].sum()
gm_data.columns = [
    r"$\mu$ [km<sup>3</sup>/s<sup>2</sup>]",
    "Mass [kg]",
    "Mass Fraction [kg/kg]",
]
gm_data
```

## Planetary Orbital Elements

The data in @tab:planetary-orbital-elements is generated by the following script, using data from {cite}`Park2021,Rhodes2019`:

```{code-cell} python
:tags: [hide-cell]
:label: code:planetary-orbital-elements
import re
from datetime import datetime
from pathlib import Path
from typing import TypedDict

import numpy as np
import pandas as pd
from dateutil import tz
from dateutil.rrule import DAILY, rrule
from numpy import typing as npt
from skyfield.data.iers import build_timescale_arrays, parse_dut1_from_finals_all
from skyfield.elementslib import OsculatingElements, osculating_elements_of
from skyfield.framelib import ecliptic_frame
from skyfield.iokit import load_file
from skyfield.timelib import Timescale
from skyfield.units import Distance, Velocity

CelestialObject = TypedDict(
    "CelestialObject",
    {
        "Celestial Body": str,
        "Semi-Major Axis (km)": float,
        "Perihelion (km)": float,
        "Aphelion (km)": float,
        "Period (days)": float,
        "Velocity (km/s)": float,
        "Eccentricity": float,
        "Inclination (degrees)": float,
    },
)


def create_celestial_object(name: str, elems: OsculatingElements) -> CelestialObject:
    return {
        "Celestial Body": name,
        "Semi-Major Axis (km)": np.average(elems.semi_major_axis.km),
        "Perihelion (km)": np.average(elems.periapsis_distance.km),
        "Aphelion (km)": np.average(elems.apoapsis_distance.km),
        "Period (days)": np.average(elems.period_in_days),
        "Velocity (km/s)": np.sqrt(elems._mu / np.average(elems.semi_major_axis.km)),
        "Eccentricity": np.average(elems.eccentricity),
        "Inclination (degrees)": np.average(elems.inclination.degrees),
    }


def parse_pck_file(filename: Path) -> dict[int | None, float]:
    m = re.compile(
        r"^\s+BODY(?P<identifier>\d+)_GM\s+=\s+\(\s*(?P<value>[\d.E+-]+)\s*\)$"
    )
    GM_dict: dict[int | None, float] = {None: 0.0}
    identifier_int = 0
    lines = filename.read_text().splitlines()
    start_processing = False
    for line in lines:
        if not line.strip():
            continue
        if "begindata" in line:
            start_processing = True
            continue
        if "begintext" in line and start_processing:
            break
        if not start_processing:
            continue
        mat = m.search(line)
        if mat is not None:
            identifier = mat.group("identifier")
            value = mat.group("value")
            if isinstance(value, str) and isinstance(identifier, str):
                identifier_int = int(identifier)
                GM_dict[identifier_int] = float(value)
            else:
                raise ValueError(value)

    return GM_dict


HERE = Path.cwd()
if HERE.name == "reference":
    HERE /= "scripts"
elif HERE.name == "orbital-mechanics-notes":
    HERE = HERE.joinpath("reference", "scripts")
GM_dict = parse_pck_file(HERE / "gm_Horizons.pck")
planets = load_file(HERE / "de442s.bsp")
with HERE.joinpath("finals2000A.all").open("rb") as f:
    utc_mjd, dut1 = parse_dut1_from_finals_all(f)
daily_tt, daily_delta_t, leap_dates, leap_offsets = build_timescale_arrays(
    utc_mjd, dut1
)
ts = Timescale((daily_tt, daily_delta_t), leap_dates, leap_offsets)

start_date = datetime(1849, 12, 26, tzinfo=tz.UTC)
end_date = datetime(2150, 1, 22, tzinfo=tz.UTC)
dates: list[datetime] = list(rrule(DAILY, dtstart=start_date, until=end_date))
t = ts.from_datetimes(dates)
ecliptic_frame_dates: npt.NDArray = ecliptic_frame.rotation_at(t)

planet_raw_data: list[CelestialObject] = []
for planet_name, center in (
    ("Mercury", "Sun"),
    ("Venus", "Sun"),
    ("Earth", "Sun"),
    ("Moon", "Earth"),
    ("Mars", "Sun"),
    ("Jupiter", "Sun"),
    ("Saturn", "Sun"),
    ("Uranus", "Sun"),
    ("Neptune", "Sun"),
    ("Pluto", "Sun"),
):
    planet = (planet_name + " Barycenter") if planet_name != "Moon" else planet_name
    position = (planets[planet] - planets[center]).at(t)

    # This is broken for vectors of times and the ecliptic frame matrix, see
    # https://github.com/skyfielders/python-skyfield/issues/655#issuecomment-960377889
    # elems = osculating_elements_of(position, ecliptic_frame)

    # So we have to do things manually
    elems = OsculatingElements(
        Distance(np.einsum("mnr,nr->mr", ecliptic_frame_dates, position.xyz.au)),
        Velocity(
            np.einsum("mnr,nr->mr", ecliptic_frame_dates, position.velocity.au_per_d)
        ),
        position.t,
        GM_dict[position.center] + GM_dict[position.target],
    )
    planet_raw_data.append(create_celestial_object(planet_name, elems))

planet_data = pd.DataFrame(planet_raw_data).set_index("Celestial Body")
planet_data
```
